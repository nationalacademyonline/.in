   function constructPDFURL(roll, dob):

       last4Roll = roll.slice(-4)
       year = dob.split('-')[0]
       last2Year = year.slice(-2)
       pdfCode = last4Roll + last2Year
       return "https://raw.githubusercontent.com/nationalacademyonline/.in/main/CER/Result_PDFs/" + pdfCode + ".pdf"
